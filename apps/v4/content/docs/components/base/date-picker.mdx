## نصب

انتخابگر تاریخ از ترکیب [`Popover`](/docs/components/base/popover) و [`Calendar`](/docs/components/base/calendar) ساخته می‌شود.

```bash
npx farsiui@latest add popover
npx farsiui@latest add calendar
```

## استفاده

```tsx showLineNumbers title="components/example-date-picker.tsx"
"use client"

import * as React from "react"
import { Calendar as CalendarIcon } from "lucide-react"

import { Button } from "@/components/ui/button"
import { Calendar } from "@/components/ui/calendar"
import {
  Popover,
  PopoverContent,
  PopoverTrigger,
} from "@/components/ui/popover"

function formatDate(date: Date) {
  return date.toLocaleDateString("fa-IR-u-ca-persian", {
    day: "numeric",
    month: "long",
    year: "numeric",
  })
}

export function DatePickerDemo() {
  const [date, setDate] = React.useState<Date>()

  return (
    <div dir="rtl">
      <Popover>
        <PopoverTrigger
          render={
            <Button
              variant="outline"
              data-empty={!date}
              className="justify-start text-start font-normal data-[empty=true]:text-muted-foreground"
            />
          }
        >
          <CalendarIcon />
          {date ? formatDate(date) : <span>انتخاب تاریخ</span>}
        </PopoverTrigger>
        <PopoverContent className="w-auto p-0">
          <Calendar mode="single" selected={date} onSelect={setDate} />
        </PopoverContent>
      </Popover>
    </div>
  )
}
```

برای جزئیات بیشتر، مستندات [React DayPicker](https://react-day-picker.js.org) را ببینید.
